<script>
    (function() {
      const form = document.getElementById('edit-regional-form');
      const banner = document.getElementById('status-banner');
      const submitBtn = document.getElementById('btn-submit');
      const cancelBtn = document.getElementById('btn-cancel');
      const cancelTopBtn = document.getElementById('btn-cancel-top');
      const nameInput = document.getElementById('reg-name');
      const codeInput = document.getElementById('reg-code');
      const activeCheckbox = document.getElementById('reg-active');
      const nameIcon = document.getElementById('name-check-icon');

      // Live validation on Name input
      nameInput.addEventListener('input', function() {
        if (this.value.trim().length > 0) {
          nameIcon.classList.remove('hidden');
          nameIcon.textContent = 'check';
          nameIcon.className = 'absolute right-3 top-3 material-symbols-outlined text-[20px] text-tertiary';
        } else {
          nameIcon.textContent = 'error';
          nameIcon.className = 'absolute right-3 top-3 material-symbols-outlined text-[20px] text-error';
        }
      });

      // Form submission simulation
      form.addEventListener('submit', function(e) {
        e.preventDefault();
        
        // Button loading state
        const originalText = submitBtn.innerHTML;
        submitBtn.disabled = true;
        submitBtn.innerHTML = '<span class="material-symbols-outlined text-[18px] animate-spin">refresh</span><span>Salvando...</span>';

        setTimeout(function() {
          submitBtn.disabled = false;
          submitBtn.innerHTML = originalText;
          banner.classList.remove('hidden');
          
          // Scroll smoothly to top if needed
          window.scrollTo({ top: 0, behavior: 'smooth' });
        }, 600);
      });

      // Cancellation triggers
      function handleCancel() {
        nameInput.value = 'São Paulo';
        codeInput.value = 'Pia-Reg-SP';
        activeCheckbox.checked = true;
        nameIcon.textContent = 'check';
        nameIcon.className = 'absolute right-3 top-3 material-symbols-outlined text-[20px] text-tertiary';
        banner.classList.add('hidden');
      }

      cancelBtn.addEventListener('click', handleCancel);
      cancelTopBtn.addEventListener('click', handleCancel);
    })();
  </script>
